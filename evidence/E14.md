    def active_skills(self) -> list[str]:
        """Currently active skill ids, in load order."""
        return list(self._active)

    @property
    def active_tool_names(self) -> set[str]:
        """Set of tool names dispatchable right now.

        Domain tools whose surface is silenced (e.g. ``ui_open_widget_editor``
        while dashboards are silenced) are filtered out even when their owning
        skill is active, so the agent cannot dispatch them. Reversible via the
        denylist — see ``silenced_domain_tool_names``.
        """
        hidden = silenced_domain_tool_names(agent_id=self.agent_id)
        names: set[str] = set()
        for sid in self._active:
            for tool in self.registry.get(sid).tools:
                if tool.name in hidden:
                    continue
                if tool.name in self.denied_tool_names:
                    continue
                names.add(tool.name)
        return names

    def can_dispatch(self, tool_name: str) -> bool:
        return tool_name in self.active_tool_names

    def is_skill_allowed(self, skill_id: str) -> bool:
        return self.allowed_skill_ids is None or skill_id in self.allowed_skill_ids

    def is_active(self, skill_id: str) -> bool:
        return skill_id in self._active

    def update_runtime_context(self, patch: dict[str, Any]) -> None:
        """Merge new keys into the snapshot.

        Raises ``RuntimeError`` in provider mode — there is no snapshot
        to mutate; update the provider's source instead.
        """
        if self._provider is not None:
            raise RuntimeError(
                "SkillSession is in provider mode; cannot update_runtime_context. "
                "Mutate the source of truth (e.g. the RuntimeContext model "
                "passed to the harness) and the session will see the change."
            )
        assert self._snapshot is not None
        self._snapshot.update(patch)
