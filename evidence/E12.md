def prepare_modeler_unified(
    payload: dict[str, Any],
    *,
    registry: SkillRegistry | None = None,
    dispatch_registry: DispatchRegistry | None = None,
    agent_id: str = "modeler",
    model: str = DEFAULT_MODELER_UNIFIED_MODEL,
    reasoning_effort: str = "high",  # see build_modeler_unified_agent
    spec_store: Any = None,  # legacy kwarg, ignored — IntentPlan retirement
    summariser: Any = None,
) -> tuple[Agent, SkillSession, DispatchRegistry, RuntimeContext]:
    """Build everything needed for one Modeler unified turn.

    Returns the constructed Agent plus the live SkillSession,
    DispatchRegistry and RuntimeContext. Useful for tests and for the
    actual run loop later. ``spec_store`` is accepted for backwards
    compat with older test fixtures; the value is no longer wired
    anywhere (IntentPlan retirement, May 2026).
    """
    del spec_store
    runtime_context = request_to_runtime_context(payload)
    agent, session, dispatch = build_modeler_unified_agent(
        runtime_context=runtime_context,
        registry=registry,
        dispatch_registry=dispatch_registry,
        agent_id=agent_id,
        model=model,
        reasoning_effort=reasoning_effort,
        summariser=summariser,
    )
    return agent, session, dispatch, runtime_context
