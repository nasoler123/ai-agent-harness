async def _apply_presend_budget_gate(
    segment_input: Any,
    *,
    budget_tokens: Optional[int],
    context: Any,
    log_tag: str,
    segment_number: int,
) -> tuple[Any, Optional[ContextWindowBudgetExceeded], int]:
    """Pre-send context-window budget gate, shared by BOTH run paths (streaming
    ``run_streamed`` and sync ``run``). Never dispatch a request we already know
    is over budget — the staging stream-cut incident assembled ~2.3M tokens
    against a 400K window and OpenAI dropped the stream mid-body.

    Cascade: estimate -> release stale tool bodies -> summarise older history ->
    fail soft. Returns ``(possibly_reduced_input, error_or_None, compactions)``.
    When ``error`` is not None the caller MUST NOT dispatch; it is a
    ``ContextWindowBudgetExceeded`` that ``run_modeler_unified`` routes to the
    friendly envelope + recovery checkpoint. ``compactions`` is how many in-run
    compaction passes ran (for the caller's counters). ``None``/0 budget
    disables the gate.

    The estimate is exact when no ``previous_response_id`` chain is active (the
    incident case); when a chain IS active it is a lower bound, so an
    over-budget estimate still correctly signals trouble.
    """
    if not budget_tokens or budget_tokens <= 0:
        return segment_input, None, 0
    estimate = _estimate_input_tokens_for_gate(segment_input, budget_tokens)
    if estimate <= budget_tokens:
        return segment_input, None, 0

    compactions = 0
    # Lever 1: release stale tool-output bodies (cheap, deterministic — Phase 1).
    if isinstance(segment_input, list) and has_releaseable_tool_outputs(segment_input):
        segment_input = compact_run_history(segment_input)
        compactions += 1
        estimate = _estimate_input_tokens_for_gate(segment_input, budget_tokens)
        logger.warning(
            "MODELER_PRESEND_BUDGET_COMPACTED%s segment=%d estimated_after=%d "
            "budget=%d",
            log_tag,
            segment_number,
            estimate,
            budget_tokens,
        )
    # Lever 2: summarise the older history into a dense recap (Phase 2A —
    # summarize-and-continue). Async, best-effort, pairing-safe; on any failure
    # the history is unchanged and we fall through to fail-soft below.
    if estimate > budget_tokens and isinstance(segment_input, list):
        summarized, did_summarize = await summarize_run_history(
            segment_input,
            conversation_id=_trace_context_value(context, "conversation_id"),
        )
        if did_summarize:
            segment_input = summarized
            compactions += 1
            estimate = _estimate_input_tokens_for_gate(segment_input, budget_tokens)
            logger.warning(
                "MODELER_PRESEND_BUDGET_SUMMARIZED%s segment=%d estimated_after=%d "
                "budget=%d",
                log_tag,
                segment_number,
                estimate,
                budget_tokens,
            )
    if estimate > budget_tokens:
        logger.warning(
            "MODELER_PRESEND_BUDGET_EXCEEDED%s segment=%d estimated_input_tokens=%d "
            "budget=%d — refusing to dispatch (fail-soft)",
            log_tag,
            segment_number,
            estimate,
            budget_tokens,
        )
        return (
            segment_input,
            ContextWindowBudgetExceeded(
                f"assembled input ~{estimate} tokens exceeds the context-window "
                f"budget of {budget_tokens} tokens after in-run compaction"
            ),
            compactions,
        )
    return segment_input, None, compactions
