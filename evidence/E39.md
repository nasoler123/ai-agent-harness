class E2BSandboxConfig:
    enabled: bool
    api_key_present: bool
    timeout_seconds: int
    max_files: int
    max_bytes: int
    mock_mode: bool = False

    @property
    def configured(self) -> bool:
        return self.enabled and (self.api_key_present or self.mock_mode)


def e2b_config_from_env() -> E2BSandboxConfig:
    # Provider lock-in/cost gate: feature flag, timeout and usage limit are
    # explicit before real E2B execution can happen.
    return E2BSandboxConfig(
        enabled=_env_flag("PIPELINES_E2B_ENABLED", False),
        api_key_present=bool(str(os.getenv("E2B_API_KEY") or "").strip()),
        timeout_seconds=int(os.getenv("PIPELINES_E2B_TIMEOUT_SECONDS") or "60"),
        max_files=int(os.getenv("PIPELINES_E2B_MAX_FILES") or "200"),
        max_bytes=int(os.getenv("PIPELINES_E2B_MAX_BYTES") or str(2 * 1024 * 1024)),
        mock_mode=_env_flag("PIPELINES_E2B_MOCK", False),
    )
