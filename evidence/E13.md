    def _validate_no_co_load_cycles(self) -> None:
        """Detect cycles in the co_loads graph using a 3-color DFS.

        Companions referenced but not present in the registry are NOT
        validated here — that's a configuration warning surfaced at
        load time, not a startup error (a skill might reference a
        sibling that lives in another agent's skill set).
        """
        WHITE, GRAY, BLACK = 0, 1, 2
        color: dict[str, int] = {sid: WHITE for sid in self._skills}

        def visit(sid: str, stack: list[str]) -> None:
            color[sid] = GRAY
            stack.append(sid)
            for companion in self._skills[sid].policies.co_load.co_loads:
                if companion not in self._skills:
                    continue  # missing companions are runtime warnings
                if color[companion] == GRAY:
                    cycle_start = stack.index(companion)
                    cycle = stack[cycle_start:] + [companion]
                    raise SkillDiscoveryError(
                        "Cycle detected in co_loads: "
                        + " -> ".join(cycle)
                    )
                if color[companion] == WHITE:
                    visit(companion, stack)
            stack.pop()
            color[sid] = BLACK

        for sid in self._skills:
            if color[sid] == WHITE:
                visit(sid, [])
